INTEGRATION STEP — do this manually, don't overwrite your existing layout.tsx

Open src/app/layout.tsx. Find where {children} is rendered inside <body>, and wrap it
with <ThemeProvider>. Also add the import at the top.

Example — if your layout currently looks like:

  import "./globals.css";

  export default function RootLayout({ children }: { children: React.ReactNode }) {
    return (
      <html lang="en">
        <body>
          {children}
        </body>
      </html>
    );
  }

Change it to:

  import "./globals.css";
  import { ThemeProvider } from "@/lib/theme";

  export default function RootLayout({ children }: { children: React.ReactNode }) {
    return (
      <html lang="en">
        <body>
          <ThemeProvider>
            {children}
          </ThemeProvider>
        </body>
      </html>
    );
  }

Everything else in your layout.tsx (metadata, fonts, etc.) stays exactly as-is —
just add the import and the <ThemeProvider> wrapper around {children}.
